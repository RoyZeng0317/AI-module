"""以 Python for 迴圈計算真正卷積；Pillow 僅負責影像讀寫。

卷積先將核旋轉 180 度，因此 Sobel 符號與直接套核的互相關相反。
灰階轉換是逐像素色彩運算，並非空間卷積。原始結果不裁切、不取整。
"""
import argparse, json
from pathlib import Path
from PIL import Image

ROOT = Path(__file__).resolve().parents[2]
MATRIX_A = [
    [250, 231, 255, 253, 251, 253, 254],
    [251, 251, 255, 253, 252, 255, 255],
    [251, 254, 250, 231, 255, 253, 255],
    [253, 252, 255, 255, 254, 250, 231],
    [255, 253, 252, 255, 255, 254, 250],
    [250, 251, 255, 253, 255, 253, 252],
    [255, 253, 251, 253, 254, 255, 255],
    [252, 255, 255, 254, 250, 253, 251],
]
FILTERS = {
    "filter1": [[1, 0, 1], [0, 4, 0], [1, 0, 1]],
    "filter2": [[1 / 25 for _ in range(5)] for _ in range(5)],
    "filter3": [[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]],
}


def convolve(matrix, kernel, padding="edge"):
    """純迴圈單通道卷積；edge/zero 保持尺寸，valid 不補邊。"""
    if not matrix or not matrix[0] or any(len(row) != len(matrix[0]) for row in matrix):
        raise ValueError("影像矩陣必須非空且為矩形")
    if (not kernel or not kernel[0] or len(kernel) % 2 == 0
            or len(kernel[0]) % 2 == 0
            or any(len(row) != len(kernel[0]) for row in kernel)):
        raise ValueError("卷積核必須非空、矩形，且長寬為奇數")
    if padding not in ("edge", "zero", "valid"):
        raise ValueError("padding 必須為 edge、zero 或 valid")
    height, width = len(matrix), len(matrix[0])
    kh, kw = len(kernel), len(kernel[0])
    ry, rx = kh // 2, kw // 2
    if padding == "valid" and (height < kh or width < kw):
        raise ValueError("valid 模式的影像不可小於卷積核")
    ys = range(ry, height - ry) if padding == "valid" else range(height)
    xs = range(rx, width - rx) if padding == "valid" else range(width)
    result = []
    for y in ys:
        row = []
        for x in xs:
            value = 0
            for ky in range(kh):
                iy = y + ky - ry
                for kx in range(kw):
                    weight = kernel[kh - 1 - ky][kw - 1 - kx]
                    if weight == 0:
                        continue
                    ix = x + kx - rx
                    if padding == "zero" and not (0 <= iy < height and 0 <= ix < width):
                        continue
                    sy, sx = max(0, min(height - 1, iy)), max(0, min(width - 1, ix))
                    value += matrix[sy][sx] * weight
            row.append(value)
        result.append(row)
    return result


def read_channels(path):
    """逐點讀 RGB；灰階輸入複製至三色，透明輸入忽略 alpha。"""
    with Image.open(path) as image:
        if image.mode not in ("L", "RGB", "RGBA"):
            raise ValueError(f"不支援 {image.mode} 模式；請提供 L、RGB 或 RGBA")
        channels = [[], [], []]
        pixels = image.load()
        for y in range(image.height):
            rows = [[], [], []]
            for x in range(image.width):
                pixel = pixels[x, y]
                for c in range(3):
                    rows[c].append(pixel if image.mode == "L" else pixel[c])
            for c in range(3):
                channels[c].append(rows[c])
    return channels


def grayscale(channels):
    result = []
    for y in range(len(channels[0])):
        row = []
        for x in range(len(channels[0][0])):
            row.append(.299 * channels[0][y][x] + .587 * channels[1][y][x]
                       + .114 * channels[2][y][x])
        result.append(row)
    return result


def save_display(channels, path, minmax=False):
    """顯示檔才四捨五入；minmax 使用所有通道共同範圍保留相對色彩。"""
    low, high = float("inf"), float("-inf")
    for channel in channels:
        for row in channel:
            for value in row:
                low, high = min(low, value), max(high, value)
    image = Image.new("L" if len(channels) == 1 else "RGB",
                      (len(channels[0][0]), len(channels[0])))
    pixels = image.load()
    for y in range(image.height):
        for x in range(image.width):
            values = []
            for channel in channels:
                value = channel[y][x]
                if minmax:
                    value = 255 * (value - low) / (high - low) if high > low else 0
                values.append(max(0, min(255, round(value))))
            pixels[x, y] = values[0] if len(values) == 1 else tuple(values)
    image.save(path)


def process_image(source, output, padding="edge"):
    output = Path(output)
    output.mkdir(parents=True, exist_ok=True)
    rgb = read_channels(source)
    gray = grayscale(rgb)
    save_display([gray], output / "gray_original.png")
    for mode, channels in (("rgb", rgb), ("gray", [gray])):
        for name, kernel in FILTERS.items():
            result = []
            for channel in channels:
                result.append(convolve(channel, kernel, padding))
            stem = output / f"{mode}_{name}"
            with stem.with_suffix(".json").open("w", encoding="utf-8") as file:
                json.dump({"padding": padding, "kernel": kernel, "channels": result}, file)
            save_display(result, output / f"{mode}_{name}_clipped.png")
            save_display(result, output / f"{mode}_{name}_minmax.png", minmax=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("inputs", type=Path, nargs="*", help="不改動來源圖片")
    parser.add_argument("--output-dir", type=Path, default=ROOT / "output/image-processing/convolution")
    parser.add_argument("--padding", choices=("edge", "zero", "valid"), default="edge")
    args = parser.parse_args()
    output = args.output_dir.resolve()
    if not output.is_relative_to(ROOT):
        parser.error("輸出資料夾必須位於本專案內")
    sources = args.inputs or [ROOT / "新手大禮包/旋積運算/Lenna_RGB.tif"]
    for source in sources:
        destination = output / source.stem
        if source.resolve().is_relative_to(destination.resolve()):
            parser.error("來源不可放在產出資料夾內，以避免覆寫")
    output.mkdir(parents=True, exist_ok=True)
    with (output / "matrix_valid.json").open("w", encoding="utf-8") as file:
        json.dump({name: convolve(MATRIX_A, kernel, "valid")
                   for name, kernel in FILTERS.items()}, file, indent=2)
    for source in sources:
        process_image(source, output / source.stem, args.padding)
        print(f"已輸出：{output / source.stem}")


if __name__ == "__main__":
    main()

"""以 for 迴圈計算 Y/Cb/Cr、Y 直方圖與均衡化；不使用 NumPy。"""
import argparse, csv, json
from pathlib import Path
from PIL import Image, ImageDraw

PROJECT_DIR = Path(__file__).resolve().parents[2]
COEFFICIENTS = ((16, 65.738, 129.057, 25.064),
                (128, -37.945, -74.494, 112.439),
                (128, 112.439, -94.154, -18.285))


def byte(value):
    return max(0, min(255, int(value + .5)))


def rgb_to_ycbcr(image):
    """練習文件指定的有限範圍公式；各通道以灰階容器呈現。"""
    if image.mode != 'RGB' or not image.width or not image.height:
        raise ValueError('輸入必須是非空 RGB 圖片')
    channels = [Image.new('L', image.size) for _ in range(3)]
    source, outputs = image.load(), [channel.load() for channel in channels]
    for y in range(image.height):
        for x in range(image.width):
            r, g, b = source[x, y]
            for pixels, (offset, cr, cg, cb) in zip(outputs, COEFFICIENTS):
                pixels[x, y] = byte(offset + (cr*r + cg*g + cb*b) / 256)
    return tuple(channels)


def histogram(image):
    if image.mode != 'L' or not image.width or not image.height:
        raise ValueError('直方圖輸入必須是非空灰階圖片')
    counts, pixels = [0] * 256, image.load()
    for y in range(image.height):
        for x in range(image.width):
            counts[pixels[x, y]] += 1
    return counts


def equalization_table(counts):
    if len(counts) != 256 or any(type(n) is not int or n < 0 for n in counts) or not sum(counts):
        raise ValueError('需要256個非負整數bin，且總像素數須大於0')
    cdf, total = [], 0
    for count in counts:
        total += count
        cdf.append(total)
    first = next(value for value in cdf if value)
    denominator = total - first
    lut = []
    for value in cdf:
        # 整數四捨五入；單一灰階時保留原圖，避免除以0。
        mapped = ((value-first)*255*2 + denominator) // (2*denominator) if denominator else len(lut)
        lut.append(max(0, min(255, mapped)))
    return cdf, lut


def apply_table(image, lut):
    if image.mode != 'L' or len(lut) != 256 or any(type(n) is not int or not 0 <= n <= 255 for n in lut):
        raise ValueError('需要灰階圖片及256個0至255的整數映射值')
    result = Image.new('L', image.size)
    source, output = image.load(), result.load()
    for y in range(image.height):
        for x in range(image.width):
            output[x, y] = lut[source[x, y]]
    return result


def histogram_chart(counts, title):
    canvas = Image.new('RGB', (576, 320), 'white')
    draw, peak = ImageDraw.Draw(canvas), max(counts)
    draw.text((32, 8), title, fill='black')
    draw.line((32, 280, 544, 280), fill='black')
    for level, count in enumerate(counts):
        height = count * 235 / peak if peak else 0
        draw.line((32+level*2, 279, 32+level*2, 279-height), fill='#3070b0', width=2)
    for level in (0, 64, 128, 192, 255):
        draw.text((32+level*2-5, 287), str(level), fill='black')
    draw.text((32, 304), f'peak={peak}; pixels={sum(counts)}', fill='black')
    return canvas


def process(source, directory):
    source, directory = Path(source).resolve(), Path(directory).resolve()
    with Image.open(source) as image:
        # RGB模式解碼，不呼叫內建灰階／YCbCr／均衡化轉換。
        rgb = image.convert('RGB')
    y, cb, cr = rgb_to_ycbcr(rgb)
    before = histogram(y)
    cdf, lut = equalization_table(before)
    equalized = apply_table(y, lut)
    after = histogram(equalized)
    directory.mkdir(parents=True, exist_ok=True)
    images = {'Y': y, 'Cb': cb, 'Cr': cr, 'Y_equalized': equalized,
              'Y_histogram': histogram_chart(before, 'Y histogram before'),
              'Y_equalized_histogram': histogram_chart(after, 'Y histogram after')}
    for name, image in images.items():
        image.save(directory / f'{source.stem}_{name}.png')
    with (directory / f'{source.stem}_histogram.csv').open('w', newline='', encoding='utf-8') as stream:
        writer = csv.writer(stream)
        writer.writerow(('gray_level', 'before_count', 'cdf_count', 'lut', 'after_count'))
        for level in range(256):
            writer.writerow((level, before[level], cdf[level], lut[level], after[level]))
    preview = Image.new('RGB', (rgb.width*3, rgb.height+24), 'white')
    draw = ImageDraw.Draw(preview)
    for column, (title, image) in enumerate((('RGB source', rgb), ('Y (limited range)', y), ('Y equalized', equalized))):
        preview.paste(image, (column*rgb.width, 24))
        draw.text((column*rgb.width+8, 5), title, fill='black')
    preview.save(directory / f'{source.stem}_comparison.png')
    report = dict(source=str(source), size=list(rgb.size), pixels=rgb.width*rgb.height,
                  y_range=[next(i for i,n in enumerate(before) if n), max(i for i,n in enumerate(before) if n)],
                  equalized_range=[next(i for i,n in enumerate(after) if n), max(i for i,n in enumerate(after) if n)],
                  mapping='CDF-min; round-half-up; constant images unchanged', coefficients=COEFFICIENTS)
    (directory / f'{source.stem}_report.json').write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding='utf-8')
    return report


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('source', nargs='?', type=Path,
                        default=PROJECT_DIR/'新手大禮包/直方圖均衡化/Lenna_RGB.tif')
    parser.add_argument('--output', type=Path, default=PROJECT_DIR/'output/image-processing/histogram')
    args = parser.parse_args()
    print(json.dumps(process(args.source, args.output), ensure_ascii=False, indent=2))

## 卷積

`convolve(matrix, kernel, padding="edge")` 使用四層for迴圈逐像素／逐核元素累加，核先翻轉180度，是真正的數學卷積。預設複製邊界而維持尺寸；`--padding zero` 補零，`--padding valid` 不補邊。

依 `新手大禮包/旋積運算/旋積運算.docx` 的三個核：

```text
Filter 1       Filter 2                 Filter 3
1 0 1          5×5全1，再乘1/25         -1  0  1
0 4 0                                   -2  0  2
1 0 1                                   -1  0  1
```

Filter 1不擅自除以8；Filter 2是平均濾波；Filter 3翻核後符號與直接套Sobel核的互相關相反。RGB各通道分開處理；手動灰階使用0.299R+0.587G+0.114B，這個色彩轉換本身不是空間卷積。

- `matrix_valid.json`：練習MatrixA的完整原始結果，Filter1/3為6×5（高×寬）、Filter2為4×3。第一列分別為 `[2010,1989,2023,1998,2035]`、`[251.28,251.48,251.64]`（顯示取至小數點後2位）、`[-12,-3,5,-26,-9]`。
- 各圖片子目錄的 `rgb_filter*.json`／`gray_filter*.json`：不裁切、不四捨五入的原始數值，保留負值及超過255的值。
- `*_clipped.png`：只在顯示時裁切至0–255；Filter1可能因此大片變白，是原核係數的結果。
- `*_minmax.png`：只在顯示時把所有通道共同值域映射至0–255，不改原始JSON或核係數；常數矩陣的min–max顯示為0。
- `gray_original.png`：手動轉灰階，沒有套空間濾波。圖片結果保留各自來源尺寸。

## Y、Cb、Cr與均衡化

RGB直接表示紅／綠／藍；YCbCr分開亮度Y與藍色色差Cb、紅色色差Cr。Cb、Cr存成灰階圖片只是查看通道值，並非兩張不同的亮度照片。此處依 `直方圖均衡化程式練習.docx` 的有限範圍公式，不用Pillow內建YCbCr轉換：

```text
Y  = 16  + (65.738R + 129.057G + 25.064B) / 256
Cb = 128 + (-37.945R - 74.494G + 112.439B) / 256
Cr = 128 + (112.439R - 94.154G - 18.285B) / 256
```

逐像素四捨五入並限制至0–255；再用for迴圈統計Y的256個bins、累積CDF，建立映射：

```text
new_y = round_half_up((CDF[y] - CDF_min) * 255 / (像素總數 - CDF_min))
```

單一灰階的圖片保留原值，避免除以0。這是全域均衡化，不是CLAHE；離散映射不保證直方圖完全平坦，也不能只憑值域變大就保證視覺效果較好。

輸出 `*_Y.png`、`*_Cb.png`、`*_Cr.png`、`*_Y_equalized.png`、前後直方圖PNG及 `*_histogram.csv`（原計數、累積計數、LUT、均衡化後計數）。`*_comparison.png` 為原RGB／Y／均衡化Y對照。

本次Lenna_RGB.webp實測：262144個像素、Y範圍34–224，均衡化後0–255。

import ast
from pathlib import Path
import pytest
from PIL import Image
from lib.opencv2.convolution import FILTERS, MATRIX_A, convolve, grayscale, read_channels, save_display


def test_true_convolution_flips_asymmetric_kernel():
    matrix = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
    assert convolve(matrix, [[1, 2, 3], [0, 0, 0], [0, 0, 0]], "valid") == [[46]]
    assert convolve(matrix, FILTERS["filter3"], "valid") == [[-8]]


def test_padding_and_raw_range():
    assert convolve([[10]], [[1] * 3 for _ in range(3)], "edge") == [[90]]
    assert convolve([[10]], [[1] * 3 for _ in range(3)], "zero") == [[10]]
    assert convolve([[250] * 3 for _ in range(3)], FILTERS["filter1"], "valid") == [[2000]]
    assert convolve([[3, 8]], [[1]]) == [[3, 8]]
    result = convolve([[10] * 5 for _ in range(5)], FILTERS["filter2"], "valid")
    assert result[0][0] == pytest.approx(10)


@pytest.mark.parametrize("kernel", [[], [[]], [[1, 2]], [[1], [2]], [[1, 2, 3], [1], [1, 2, 3]]])
def test_invalid_kernel(kernel):
    with pytest.raises(ValueError):
        convolve([[1]], kernel)


def test_invalid_inputs():
    for matrix in ([], [[]], [[1], [2, 3]]):
        with pytest.raises(ValueError):
            convolve(matrix, [[1]])
    with pytest.raises(ValueError):
        convolve([[1]], FILTERS["filter1"], "valid")
    with pytest.raises(ValueError):
        convolve([[1]], [[1]], "reflect")


def test_worksheet_dimensions():
    for name, expected in (("filter1", (6, 5)), ("filter2", (4, 3)), ("filter3", (6, 5))):
        result = convolve(MATRIX_A, FILTERS[name], "valid")
        assert (len(result), len(result[0])) == expected


def test_rgb_separate_channels_and_gray(tmp_path):
    source = tmp_path / "source.png"
    Image.new("RGB", (1, 1), (20, 80, 200)).save(source)
    channels = read_channels(source)
    result = [convolve(channel, [[2]]) for channel in channels]
    assert result == [[[40]], [[160]], [[400]]]
    assert grayscale(channels)[0][0] == pytest.approx(.299 * 20 + .587 * 80 + .114 * 200)
    display = tmp_path / "display.png"
    save_display(result, display)
    with Image.open(display) as image:
        assert image.getpixel((0, 0)) == (40, 160, 255)
    assert result[2][0][0] == 400

"""手算像素、CDF與常數邊界；不匯入NumPy或啟動模型。"""
import ast, csv
from pathlib import Path
import pytest
from PIL import Image
from lib.opencv2 import histogram_equalization as processing


def gray(values):
    image = Image.new('L', (len(values), 1))
    for x, value in enumerate(values):
        image.putpixel((x, 0), value)
    return image


def test_ycbcr_worksheet_known_pixels():
    image = Image.new('RGB', (5, 1))
    for x, color in enumerate(((0,0,0), (255,255,255), (255,0,0), (0,255,0), (0,0,255))):
        image.putpixel((x,0), color)
    y, cb, cr = processing.rgb_to_ycbcr(image)
    assert [y.getpixel((x,0)) for x in range(5)] == [16,235,81,145,41]
    assert [cb.getpixel((x,0)) for x in range(5)] == [128,128,90,54,240]
    assert [cr.getpixel((x,0)) for x in range(5)] == [128,128,240,34,110]


def test_hand_calculated_histogram_cdf_and_mapping():
    image = gray([10,10,20,30])
    counts = processing.histogram(image)
    cdf, lut = processing.equalization_table(counts)
    assert sum(counts) == 4 and [counts[i] for i in (10,20,30)] == [2,1,1]
    assert [cdf[i] for i in (10,20,30)] == [2,3,4]
    assert [lut[i] for i in (10,20,30)] == [0,128,255]
    result = processing.apply_table(image, lut)
    assert [result.getpixel((x,0)) for x in range(4)] == [0,0,128,255]
    assert [image.getpixel((x,0)) for x in range(4)] == [10,10,20,30]
    assert all(a <= b for a,b in zip(lut,lut[1:]))

def test_constant_image_is_unchanged(value):
